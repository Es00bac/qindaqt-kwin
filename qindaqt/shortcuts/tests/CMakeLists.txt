cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(QindaQtShortcutContracts LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_AUTOMOC ON)
    find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets DBus Test)
    find_package(KF6Config REQUIRED)
    find_package(KF6GlobalAccel REQUIRED)
    enable_testing()
    get_filename_component(FORK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
    add_library(qindaqt_kwin_shortcuts STATIC ${FORK_ROOT}/src/shortcuts/shortcut_registry.cpp)
    target_include_directories(qindaqt_kwin_shortcuts PUBLIC ${FORK_ROOT}/src)
    target_link_libraries(qindaqt_kwin_shortcuts PUBLIC Qt6::Core Qt6::Gui)
    add_library(qindaqt_kwin_shortcut_endpoint STATIC
        ../compatibility_endpoint.cpp ../compatibility_component.cpp ../compatibility_endpoint_wire.cpp
        ../compatibility_lifecycle.cpp ../native_endpoint.cpp ../runtime_store.cpp ../input_dispatcher.cpp)
    target_include_directories(qindaqt_kwin_shortcut_endpoint PUBLIC ${FORK_ROOT} ${FORK_ROOT}/src)
    target_link_libraries(qindaqt_kwin_shortcut_endpoint PUBLIC qindaqt_kwin_shortcuts Qt6::DBus KF6::ConfigCore)
else()
    set(FORK_ROOT ${CMAKE_SOURCE_DIR})
endif()
add_executable(testShortcutCompatibilityEndpoint test_shortcut_compatibility_endpoint.cpp)
target_link_libraries(testShortcutCompatibilityEndpoint PRIVATE qindaqt_kwin_shortcut_endpoint KF6::GlobalAccel Qt6::Test Qt6::Widgets)
add_test(NAME qindaqt-shortcut-compatibility COMMAND dbus-run-session -- $<TARGET_FILE:testShortcutCompatibilityEndpoint>)
set_tests_properties(qindaqt-shortcut-compatibility PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(testShortcutRuntime test_shortcut_runtime.cpp)
target_link_libraries(testShortcutRuntime PRIVATE qindaqt_kwin_shortcut_endpoint Qt6::Test)
add_test(NAME qindaqt-shortcut-runtime COMMAND $<TARGET_FILE:testShortcutRuntime>)
add_executable(testShortcutRegistry ${FORK_ROOT}/autotests/test_shortcut_registry.cpp)
target_link_libraries(testShortcutRegistry PRIVATE qindaqt_kwin_shortcuts Qt6::Test)
add_test(NAME qindaqt-shortcut-registry COMMAND $<TARGET_FILE:testShortcutRegistry>)
